using UnityEngine;
using System.Collections;

public class PlayerBase : MonoBehaviour
{

    private GameWorld gameWorld;              //Reference to world object
    private AudioSource gameAudio;             //Reference to global audio source

    private PlayerMovement playerMovement;

    void Start()
    {
        GameObject g = GameObject.Find("Game");
        gameWorld = g.GetComponent<GameWorld>();
        gameAudio = g.GetComponent<AudioSource>();

        GameObject player = GameObject.Find("Player");
        playerMovement = player.GetComponent<PlayerMovement>();
    }

    void Update()
    {
        if (gameWorld.isPaused)
            return;


    }

    public void Reset()
    {
        playerMovement.Reset();
    }
}
